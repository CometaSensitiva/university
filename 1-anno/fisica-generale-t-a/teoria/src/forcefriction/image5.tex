    \draw[force] (O) --++ (90:{\F*cos(\ang)}) node[below=2,right=0] {$\vbF_\mathrm{N}$};
    \draw[force] (O) --++ (0:{\F*sin(\ang)}) node[right=0] {$\vbF_\mathrm{a}$};
